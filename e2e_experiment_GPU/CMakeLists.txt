cmake_minimum_required(VERSION 3.24)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 80 90)
endif()
project(FELID_E2E LANGUAGES CXX CUDA)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
add_subdirectory(GPU_core)
file(GLOB FELID_RBOOT_GPU_SOURCES CONFIGURE_DEPENDS RBOOT_external/GPU/*.cu)
target_sources(felid_gpu PRIVATE ${FELID_RBOOT_GPU_SOURCES})
add_executable(felid_e2e src/main.cu src/include/source/fhe_transformer/fhe_level_utils.cu)
set_target_properties(felid_e2e PROPERTIES CUDA_SEPARABLE_COMPILATION ON LINKER_LANGUAGE CXX)
target_include_directories(felid_e2e PRIVATE src/include RBOOT_external/GPU)
target_compile_definitions(felid_e2e PRIVATE FELID_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/src/data")
target_link_libraries(felid_e2e PRIVATE -Wl,--whole-archive felid_gpu -Wl,--no-whole-archive)
